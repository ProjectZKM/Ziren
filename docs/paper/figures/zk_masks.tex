% ------------------------------------------------------------ (a) a masked sumcheck
\node[panel] at (0,4.3) {(a) A masked sumcheck on $P + \rho\, g$};
\node[rndb, anchor=west] (g) at (0.2,3.45) {$g(\bm x) = \sum_i g_i(x_i)$, $g_i(0) + g_i(1) = 0$, so $\sum_{\bm x} g = 0$};
\node[lab, anchor=west, align=left] at (0.2,2.95) {committed in the mask region of $R$ before $\rho$ is drawn};
\node[msg] (m1) at (0.9,2.3) {$m_1(X)$};
\node[msg] (m2) at (2.3,2.3) {$m_2(X)$};
\node[lab] at (3.3,2.3) {$\cdots$};
\node[msg] (mn) at (4.3,2.3) {$m_m(X)$};
\draw[a] (m1) -- node[above] {$r_1$} (m2);
\draw[a] (mn.west) ++(-0.55,0) -- node[above] {$r_{m-1}$} (mn);
\node[lab, anchor=west, align=left] at (0.2,1.45)
  {$m_v(X) = p_v(X) + \rho\,2^{f}\big(g_v(X) + \textstyle\sum_{i<v} g_i(r_i)\big)$,\\
   $f$ free variables left; $m_v(0) + m_v(1) = m_{v-1}(r_{v-1})$ holds;\\
   every non-constant coefficient is uniform.};
\node[side, anchor=north west] at (0.2,0.6)
  {Residue $m_m(r_m) - P(\bm r) = \rho\, g(\bm r)$: a linear functional of the\\
   committed cells, one cell claim.  A $g$ with cube sum $s \ne 0$ shifts\\
   the claim by $\rho s$, caught w.p. $1 - 1/|\EF|$.};

% ------------------------------------------------------------ (b) the masked LogUp-GKR output
\node[panel] at (9.0,4.3) {(b) The hidden \LogUp{} output};
\foreach \i/\t in {0/{n_0/d_0}, 1/{n_1/d_1}, 2/{\cdots}, 3/{n_{c\text{-}1}/d_{c\text{-}1}}} {

  \draw[cell] (\x,3.2) rectangle ++(1.05,0.45);
  \node[lab] at (\x+0.525,3.425) {$\t$};
}
\foreach \i/\t in {0/{p/1}, 1/{-p/1}, 2/{0/u_0}, 3/{0/u_1}} {
  \pgfmathsetmacro{\x}{9.1+1.15*(\i+4)}
  \draw[mask] (\x,3.2) rectangle ++(1.05,0.45);
  \node[lab] at (\x+0.525,3.425) {$\t$};
}
\draw[zirenslate, decorate, decoration={brace, amplitude=3pt, mirror}]
  (9.1,3.1) -- node[lab, below=3pt] {real output $\mathsf{out}$} (9.1+1.15*4-0.1,3.1);
\draw[zirenslate, decorate, decoration={brace, amplitude=3pt, mirror}]
  (9.1+1.15*4,3.1) -- node[lab, below=3pt] {virtual slice, $\sum = 0$} (9.1+1.15*8-0.1,3.1);
\node[lab, anchor=north west, align=left] at (9.0,2.5)
  {Two merge layers over $\mathsf{out}' = (\mathsf{out}, \text{virtual})$ reach a top of two fractions,\\
   uniform given the public-value balance; $\sum \mathsf{out}' = \sum \mathsf{out}$.};
\node[side, anchor=north west] at (9.0,1.6)
  {Each layer's children carry $\tilde n_h = n_h + Z(r'_0)\,\alpha_h$, $\tilde d_h = d_h + Z(r'_0)\,\beta_h$,\\
   $Z(X) = X(1-X)$: zero on the cube, so the next sumcheck starts from a true sum;\\
   the last message adds $s\,\eqf(\bm r,(X,\bm r'))\,[Z(X)L(X) + Z(X)^2 Q]$ to match them.};

\node[side, anchor=north west] at (0.2,-2.55)
