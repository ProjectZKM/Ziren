% Random rows of a zk chip: four cancelling pairs past the real rows, the
% cut weight of the zerocheck, and the two openings of a column.
\begin{tikzpicture}[
  font=\scriptsize,
  real/.style={draw=zirenblue!90, fill=zirenblue!18, line width=0.5pt},
  rnd/.style={draw=zirenslate!80, fill=white, pattern=north east lines,
              pattern color=zirenslate!60, line width=0.5pt},
  pad/.style={draw=zirenslate!55, densely dotted, line width=0.4pt},
  wbar/.style={draw=zirenorange!90, fill=zirenorange!25, line width=0.5pt},
  lab/.style={text=black!90},
  panel/.style={font=\footnotesize\bfseries, text=black!90, anchor=west, inner sep=0pt},
  a/.style={-{Latex[length=1.6mm]}, line width=0.5pt, zirenslate!85},
  side/.style={rectangle, rounded corners=2pt, draw=zirenslate!45, fill=zirenlight,
               align=left, font=\scriptsize, inner sep=4pt, line width=0.5pt},
]

% ------------------------------------------------------------ (a) the committed matrix
\node[panel] at (0,4.25) {(a) A chip's main trace, $H = h + 8$ committed rows};
\def\W{2.6}
\def\RH{0.3}
% real rows (compressed)
\draw[real] (0.3,1.3) rectangle ++(\W,2.4);
\node[lab] at (0.3+0.5*\W,2.5) {real rows $0,\ldots,h{-}1$};
% four random pairs
\foreach \p in {0,1,2,3} {
  \pgfmathsetmacro{\y}{1.3-2*\RH*(\p+1)}
  \draw[rnd] (0.3,\y) rectangle ++(\W,{2*\RH});
  \draw[zirenslate!70, line width=0.3pt] (0.3,\y+\RH) -- ++(\W,0);
  \node[lab, anchor=west] at (0.3+\W+0.1,\y+1.5*\RH) {$\bm x_{\p}$};
  \node[lab, anchor=west] at (0.3+\W+0.1,\y+0.5*\RH) {$\bm x'_{\p}$};
}
% padded rows above H (virtual)
\draw[pad] (0.3,-1.1) rectangle ++(\W,-0.5);
\node[lab] at (0.3+0.5*\W,-1.35) {implicit zeros to $2^m$};
\draw[zirenslate, |-|] (0.1,1.3) -- node[lab, left] {$h$} (0.1,3.7);
\draw[zirenslate, |-|] (0.1,-1.1) -- node[lab, left] {$8$} (0.1,1.3);

% pair rule
\node[side, anchor=north west] at (3.9,4.0)
  {A pair $(\bm x, \bm x')$, $\bm s = \bm x + \bm x'$, $\bm d = \bm x' - \bm x$, cancels in \LogUp{}\\
   for every challenge iff, for each lookup $j$ with $L_j \ne 0$,\\[2pt]
   \quad $L_j\,\bm s = -2c_j$ \hfill (opposite multiplicities $\mu_j = c_j + L_j\,\bm x$)\\
   \quad $V_{j}\,\bm d = 0$ \hfill (equal values $v_{j} = e_{j} + V_{j}\,\bm x$)\\[2pt]
   $\bm s$, $\bm d$ are drawn uniformly from the two solution spaces;\\
   a functional $\bm a\cdot\bm x$ stays exposed iff $\bm a \in \mathrm{rowspan}(L)\cap\mathrm{rowspan}(V)$,\\
   which is $\{0\}$ for every shrink chip.  Lookups with $L_j = 0$ are\\
   switched off on the random rows and reconstructed with $h$.};

% ------------------------------------------------------------ (b) the cut weight
\node[panel] at (3.9,1.05) {(b) The zerocheck's cut weight $W(b) = \eqf(\zeta, b)\,[\,b < h\,]$};
\draw[zirenslate!70] (4.0,-0.6) -- (10.0,-0.6);
\foreach \i/\v in {0/0.55, 1/0.9, 2/0.35, 3/1.05, 4/0.7, 5/0.45, 6/0.95, 7/0.6} {
  \pgfmathsetmacro{\x}{4.1+0.42*\i}
  \draw[wbar] (\x,-0.6) rectangle ++(0.3,\v);
}
\foreach \i in {8,9,10,11} {
  \pgfmathsetmacro{\x}{4.1+0.42*\i}
  \draw[pad] (\x,-0.6) rectangle ++(0.3,0.5);
}
\draw[zirenslate, dashed] (4.1+0.42*8-0.06,-0.75) -- ++(0,1.5);
\node[lab, anchor=west] at (4.1+0.42*8,0.65) {$b = h$: zero on the random and padded rows};
\node[lab, anchor=north west, align=left] at (3.9,-0.85)
  {$\sum_b W(b)\,C(T(b)) = 0$ iff $C$ vanishes on every real row; $\widetilde W(\bm z^\ast)$ costs the verifier $O(m)$ by the bits of $h$.};

% ------------------------------------------------------------ (c) the openings
\node[side, anchor=north west] at (3.9,-1.55)
  {Each column is opened twice: $T_i(\zeta) = \sum_{b<h}\eqf(\zeta,b)\,T_i(b) + \sum_{b\in[h,H)}\eqf(\zeta,b)\,r_i(b)$ and the same at $\bm z^\ast$;\\
   the four pairs' free draws make the two sums over the random rows uniform in $\EF^2$, independent of the real cells.};
\end{tikzpicture}
